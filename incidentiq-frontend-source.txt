import { Navigate, Outlet, useLocation } from "react-router-dom";
import { useAuth } from "../../context/AuthContext";

export default function ProtectedRoute() {
  const { isAuthenticated } = useAuth();
  const location = useLocation();

  if (!isAuthenticated) {
    return (
      <Navigate
        to="/login"
        replace
        state={{ from: location }}
      />
    );
  }

  return <Outlet />;
}

function formatDate(value) {
  const date = new Date(value);

  return Number.isNaN(date.getTime())
    ? "Not available"
    : date.toLocaleString();
}

function StatusBadge({ status }) {
  const style =
    status === "completed"
      ? "border-success/25 bg-success/10 text-success"
      : "border-line bg-elevated text-text-secondary";

  return (
    <span
      className={`inline-flex whitespace-nowrap rounded border px-2 py-1 text-xs ${style}`}
    >
      {status}
    </span>
  );
}

  const [searchInput, setSearchInput] = useState("");
  const [filters, setFilters] = useState({
    search: "",
    status: "",
    page: 0,

  const [data, setData] = useState({
    items: [],
    total: 0,
  });

  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [refresh, setRefresh] = useState(0);

  useEffect(() => {
    const controller = new AbortController();

    setLoading(true);
    setError("");

    incidentService
      .list({
        search: filters.search,
        status: filters.status,
        limit: PAGE_SIZE,
        offset: filters.page * PAGE_SIZE,
        signal: controller.signal,
      })
      .then((response) => {
        if (controller.signal.aborted) return;

        setData(response);

        const lastPage = Math.max(
          0,
          Math.ceil(response.total / PAGE_SIZE) - 1
        );

        if (filters.page > lastPage) {
          setFilters((previous) => ({
            ...previous,
            page: lastPage,
          }));
        }
      })
      .catch((requestError) => {
        if (!controller.signal.aborted) {
          setError(getApiErrorMessage(requestError));
        }
      })
      .finally(() => {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      });

    return () => controller.abort();
  }, [filters, refresh]);

  function handleSearch(event) {
    event.preventDefault();

    setFilters((previous) => ({
      ...previous,
      search: searchInput.trim(),
      page: 0,
    }));
  }

  const totalPages = Math.max(
    1,
    Math.ceil(data.total / PAGE_SIZE)
  );

  const firstItem = data.total
    ? filters.page * PAGE_SIZE + 1
    : 0;

  const lastItem = Math.min(
    (filters.page + 1) * PAGE_SIZE,
    data.total
  );

  return (
    <section aria-labelledby="incident-browser-heading" className="mt-9">
      <div className="flex flex-wrap items-center justify-between gap-4">
        <div>

          >
            Saved incidents

            Shared workspace · Newest first
          </p>
        </div>

        <button
          type="button"
          onClick={() => setRefresh((value) => value + 1)}
          disabled={loading}
          className="flex h-9 items-center gap-2 rounded-md border border-line px-3 text-xs text-text-secondary hover:bg-elevated disabled:opacity-50"

          <RefreshCw size={14} aria-hidden="true" />

            <Search

            />

            <input

function CopyButton({ text, label = "Copy" }) {
  const [copied, setCopied] = useState(false);
  const [failed, setFailed] = useState(false);

  async function handleCopy() {
    try {
      await navigator.clipboard.writeText(text);
      setCopied(true);
      setFailed(false);
    } catch {
      setFailed(true);
    }
  }

  return (
    <div className="flex shrink-0 items-center gap-2">
      <button
        type="button"
        onClick={handleCopy}
        onBlur={() => {
          setCopied(false);
          setFailed(false);
        }}
        aria-label={copied ? "Copied" : label}
        title={copied ? "Copied" : label}
        className="flex h-8 items-center gap-2 rounded-md border border-line px-2 text-xs text-text-secondary hover:bg-elevated focus-visible:outline focus-visible:outline-2 focus-visible:outline-accent"
      >
        {copied ? (
          <Check size={14} aria-hidden="true" />
        ) : (
          <Copy size={14} aria-hidden="true" />
        )}
        {copied ? "Copied" : "Copy"}
      </button>

      <span role="status" className="text-xs text-text-muted">
        {failed ? "Copy unavailable" : ""}
      </span>
    </div>
  );
}

export default function InvestigationWorkspace({ incident }) {
  const [result, setResult] = useState(incident.investigation || null);
  const [running, setRunning] = useState(false);
  const [error, setError] = useState("");

  const requestInFlight = useRef(false);

  async function runInvestigation() {
    if (requestInFlight.current) return;

    requestInFlight.current = true;
    setRunning(true);
    setError("");

    try {
      const response =
        await investigationService.investigate(incident.id);

      if (
        response?.incident_id !== incident.id ||
        !Array.isArray(response?.hypotheses)
      ) {
        setError(
          "The API returned an unexpected investigation response. Reopen the incident to check its saved result."
        );
        return;
      }

      setResult(response);
    } catch (requestError) {
      setError(getApiErrorMessage(requestError));
    } finally {

import { useEffect, useState } from "react";
import { LoaderCircle, ShieldCheck } from "lucide-react";
import { useAuth } from "../../context/AuthContext";
import privacyService from "../../services/privacyService";
import { getApiErrorMessage } from "../../services/api";

export default function PrivacyControl({
  onPolicyChange,
  disabled = false,
}) {
  const { isAdmin } = useAuth();

  const [policy, setPolicy] = useState(null);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");
  const [attempt, setAttempt] = useState(0);

  useEffect(() => {
    const controller = new AbortController();

    setLoading(true);
    setError("");

    privacyService
      .get({ signal: controller.signal })
      .then((response) => {
        if (controller.signal.aborted) return;

        setPolicy(response);
        onPolicyChange?.(response);
      })
      .catch((requestError) => {
        if (!controller.signal.aborted) {
          setError(getApiErrorMessage(requestError));
          onPolicyChange?.(null);
        }
      })
      .finally(() => {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      });

    return () => controller.abort();
  }, [attempt, onPolicyChange]);

  async function togglePrivacy() {
    if (!policy || saving || disabled || !isAdmin) return;

    const nextEnabled = !policy.enabled;

    const confirmed = nextEnabled
      ? false
      : window.confirm(
          "Disable privacy mode for the entire workspace?\n\n" +
            "Future cloud evaluations may send incident logs and " +
            "retrieved operational knowledge to external providers.\n\n" +
            "No evaluation will run automatically."
        );

    if (!nextEnabled && !confirmed) return;

    setSaving(true);
    setError("");

    try {
      const updated = await privacyService.update({
        enabled: nextEnabled,
        confirmed,
      });

      setPolicy(updated);
      onPolicyChange?.(updated);
    } catch (requestError) {
      setError(getApiErrorMessage(requestError));

      // Do not keep presenting an old policy as authoritative
      // after an ambiguous failed update.
      setPolicy(null);
      onPolicyChange?.(null);
    } finally {
      setSaving(false);
    }
  }

  return (
    <section className="mt-6 rounded-lg border border-line bg-surface p-5">
      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
        <div>
          <h2 className="flex items-center gap-2 text-sm font-semibold">
            <ShieldCheck
              size={18}
              className="text-accent"
              aria-hidden="true"
            />
            Workspace privacy
          </h2>

          <p className="mt-2 text-xs leading-6 text-text-secondary">
            {loading
              ? "Loading privacy policy…"
              : !policy
                ? "Privacy policy unavailable."
                : policy.enabled
                  ? "Local processing only. Cloud evaluations are blocked."
                  : "Cloud evaluations are permitted when explicitly selected."}
          </p>

          {policy?.locked && (
            <p className="mt-1 text-xs text-warning">
              Locked by deployment configuration.
            </p>
          )}
        </div>

        {isAdmin && policy && (
          <button
            type="button"
            role="switch"
            aria-checked={policy.enabled}
            aria-label="Workspace privacy mode"
            onClick={togglePrivacy}
            disabled={saving || disabled || policy.locked}
            className="flex h-10 shrink-0 items-center gap-2 rounded-md border border-line bg-elevated px-4 text-sm disabled:cursor-not-allowed disabled:opacity-50"
          >
            {saving && (
              <LoaderCircle
                size={15}
                className="animate-spin"
                aria-hidden="true"
              />
            )}
            Privacy mode: {policy.enabled ? "On" : "Off"}
          </button>
        )}
      </div>

      {error && (
        <div className="mt-3">
          <p role="alert" className="text-sm text-red-300">
            {error}
          </p>
          <button
            type="button"
            disabled={disabled || saving}
            onClick={() => setAttempt((value) => value + 1)}
            className="mt-2 text-xs text-accent hover:underline"
          >
            Reload privacy policy
          </button>
        </div>
      )}

      <p className="mt-3 text-xs leading-5 text-text-muted">
        Changes apply to future permission checks. They do not cancel
        provider calls already in progress.
      </p>
    </section>
  );
}

import {
  createContext,
  useCallback,
  useContext,
  useMemo,
  useState,
} from "react";
import authService from "../services/authService";
import { setAccessToken } from "../services/api";

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);

  const logout = useCallback(() => {
    setAccessToken(null);
    setUser(null);
  }, []);

  const login = useCallback(async ({ username, password }) => {
    setAccessToken(null);
    setUser(null);

    try {
      const result = await authService.login({ username, password });

      if (
        typeof result.access_token !== "string" ||
        !result.access_token
      ) {
        throw new Error("The API did not return a valid access token.");
      }

      setAccessToken(result.access_token);

      const currentUser = await authService.getCurrentUser();

      if (!currentUser?.id || !currentUser?.username || !currentUser?.role) {
        throw new Error("The API returned an incomplete user profile.");
      }

      setUser(currentUser);
      return currentUser;
    } catch (error) {
      setAccessToken(null);
      setUser(null);
      throw error;
    }
  }, []);

  const register = useCallback(({ username, password }) => {
    return authService.register({ username, password });
  }, []);

  const value = useMemo(
    () => ({
      user,
      isAuthenticated: Boolean(user),
      isAdmin: user?.role === "admin",
      login,
      register,
      logout,
    }),
    [user, login, register, logout]
  );

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);

  if (!context) {
    throw new Error("useAuth must be used inside AuthProvider.");
  }

  return context;
}

export default function Incidents() {
  const navigate = useNavigate();

  const [file, setFile] = useState(null);
  const [incidentId, setIncidentId] = useState("");
  const [uploading, setUploading] = useState(false);
  const [uploadError, setUploadError] = useState("");
  const [lookupError, setLookupError] = useState("");

  async function handleUpload(event) {
    event.preventDefault();
    if (uploading) return;

    setUploadError("");

    if (!file) {
      setUploadError("Select a log file to upload.");
      return;
    }

    if (!/\.(log|txt)$/i.test(file.name)) {
      setUploadError("Only .log and .txt files are allowed.");
      return;
    }

    if (file.size > 1_000_000) {
      setUploadError("File must be 1 MB or smaller.");
      return;
    }

    setUploading(true);

    try {
      const incident = await incidentService.create(file);

      if (!incident?.id) {
        setUploadError("The API did not return an incident ID.");
        return;
      }

      navigate(`/incidents/${incident.id}`);
    } catch (error) {
      setUploadError(getApiErrorMessage(error));
    } finally {
      setUploading(false);
    }
  }

  function handleLookup(event) {
    event.preventDefault();
    setLookupError("");

    const value = incidentId.trim();

    if (!uuidPattern.test(value)) {
      setLookupError("Enter a valid incident UUID.");
      return;
    }

    navigate(`/incidents/${value}`);
  }

  return (
    <div>
      <h1 className="text-2xl font-semibold tracking-tight">

        <section className="rounded-lg border border-line bg-surface p-6">

            <h2 className="text-base font-semibold">Create an incident</h2>
          </div>

          <p className="mt-3 text-sm leading-6 text-text-secondary">
            Upload a UTF-8 log file. Incident details are generated from

function percentage(value) {
  return typeof value === "number"
    ? `${(value * 100).toFixed(1)}%`
    : "Not returned";

function ProviderResult({ item }) {
  const completed = item.status === "completed";

  const statusStyle =
    item.status === "completed"
      ? "border-success/25 bg-success/5 text-success"
      : item.status === "blocked"
        ? "border-warning/25 bg-warning/5 text-warning"
        : "border-critical/25 bg-critical/5 text-red-300";

  const providerLabel =
    PROVIDERS.find((provider) => provider.id === item.provider)?.label ||
    item.provider;

  const hypotheses = Array.isArray(item.result?.hypotheses)
    ? [...item.result.hypotheses].sort((a, b) => a.rank - b.rank)
    : [];

  const metrics = [
    [
      "Provider latency",
      typeof item.latency_ms === "number"
        ? `${item.latency_ms.toLocaleString()} ms`
        : "Not returned",
    ],
    ["Evidence coverage", percentage(item.evidence_coverage)],
    [
      "Heuristic diagnosis score",
      typeof item.diagnosis_quality === "number"
        ? `${item.diagnosis_quality.toFixed(2)} / 1`
        : "Not returned",
    ],
    ["Hypotheses", item.hypothesis_count ?? "Not returned"],
    [
      "Schema validation",
      item.schema_valid === true
        ? "Passed"
        : item.schema_valid === false
          ? "Failed"
          : "Not returned",
    ],
  ];

  return (
    <article className="min-w-0 overflow-hidden rounded-lg border border-line bg-surface">
      <header className="flex flex-wrap items-center justify-between gap-3 border-b border-line px-5 py-4">
        <h3 className="text-base font-semibold">{providerLabel}</h3>

        <span
          className={`rounded border px-2 py-1 text-xs ${statusStyle}`}
        >
          {item.status}
        </span>
      </header>

      {!completed ? (
        <div className="p-5">
          <p className="whitespace-pre-wrap break-words text-sm leading-6 text-text-secondary [overflow-wrap:anywhere]">
            {item.error || "This provider did not return a result."}
          </p>
        </div>
      ) : (
        <div className="p-5">
          <dl className="grid grid-cols-2 gap-5 border-b border-line pb-5">
            {metrics.map(([label, value]) => (
              <div key={label} className="min-w-0">
                <dt className="text-xs leading-5 text-text-muted">
                  {label}
                </dt>
                <dd className="mt-1 break-words text-sm font-medium">
                  {value}
                </dd>
              </div>
            ))}
          </dl>

          <div className="mt-5 space-y-7">
            {hypotheses.length ? (
              hypotheses.map((hypothesis, index) => {
                const evidence = Array.isArray(hypothesis.evidence)
                  ? hypothesis.evidence
                  : [];

                const suggestedChecks = Array.isArray(
                  hypothesis.suggested_checks
                )
                  ? hypothesis.suggested_checks
                  : [];

                return (
                  <section key={`${hypothesis.rank}-${index}`}>
                    <div className="flex flex-wrap items-center justify-between gap-2">
                      <h4 className="text-xs font-semibold uppercase tracking-wider text-accent">
                        Hypothesis {hypothesis.rank}
                      </h4>

                      {typeof hypothesis.confidence === "number" && (
                        <span className="text-xs text-text-muted">
                          Model confidence:{" "}
                          {percentage(hypothesis.confidence)}
                        </span>
                      )}
                    </div>

                    <p className="mt-3 whitespace-pre-wrap break-words text-sm leading-6 [overflow-wrap:anywhere]">
                      {hypothesis.summary}
                    </p>

                    <h5 className="mt-4 text-xs font-medium text-text-muted">
                      Supporting evidence
                    </h5>

                    {evidence.length ? (
                      <ul className="mt-2 space-y-3">
                        {evidence.map((reference, evidenceIndex) => (
                          <li
                            key={evidenceIndex}
                            className="border-l-2 border-line pl-3"
                          >
                            <p className="break-words text-xs leading-5 text-text-muted">
                              {reference.source_type}

                              {reference.line_number != null &&
                                ` · Line ${reference.line_number}`}

                              {reference.source_file &&
                                ` · ${reference.source_file}`}

                              {reference.chunk_index != null &&
                                ` · Chunk ${reference.chunk_index}`}
                            </p>

                            <p className="mt-1 whitespace-pre-wrap break-words text-xs leading-6 text-text-secondary [overflow-wrap:anywhere]">
                              {reference.message}
                            </p>
                          </li>
                        ))}
                      </ul>
                    ) : (
                      <p className="mt-2 text-xs text-text-secondary">
                        No evidence references returned.
                      </p>
                    )}

                    <h5 className="mt-4 text-xs font-medium text-text-muted">
                      Suggested checks
                    </h5>

                    {suggestedChecks.length ? (
                      <ol className="mt-2 list-decimal space-y-2 pl-5 text-sm leading-6 text-text-secondary">
                        {suggestedChecks.map((check, checkIndex) => (
                          <li
                            key={checkIndex}
                            className="whitespace-pre-wrap break-words [overflow-wrap:anywhere]"
                          >
                            {check}
                          </li>
                        ))}
                      </ol>
                    ) : (
                      <p className="mt-2 text-xs text-text-secondary">
                        No suggested checks returned.
                      </p>
                    )}
                  </section>
                );
              })
            ) : (
              <p className="text-sm text-text-secondary">
                No hypotheses returned.
              </p>
            )}
          </div>
        </div>
      )}
    </article>
  );
}

function ComparisonForm() {
  const [searchInput, setSearchInput] = useState("");
  const [search, setSearch] = useState("");
  const [page, setPage] = useState(0);

  const [incidents, setIncidents] = useState([]);
  const [total, setTotal] = useState(0);
  const [loadingIncidents, setLoadingIncidents] = useState(true);
  const [listError, setListError] = useState("");
  const [refresh, setRefresh] = useState(0);

  const [selectedIncident, setSelectedIncident] = useState(null);
  const [selectedProviders, setSelectedProviders] = useState([
    "ollama",
    "groq",
  ]);

  const [privacyPolicy, setPrivacyPolicy] = useState(null);

  const [running, setRunning] = useState(false);
  const [error, setError] = useState("");
  const [evaluation, setEvaluation] = useState(null);

  const inFlight = useRef(false);

  const allowedProviders = Array.isArray(
    privacyPolicy?.allowed_providers
  )
    ? privacyPolicy.allowed_providers
    : [];

  const effectiveProviders = selectedProviders.filter((provider) =>
    allowedProviders.includes(provider)
  );

  const totalPages = Math.max(1, Math.ceil(total / PAGE_SIZE));

  useEffect(() => {
    const controller = new AbortController();

    setLoadingIncidents(true);
    setListError("");

    incidentService
      .list({
        search,
        limit: PAGE_SIZE,
        offset: page * PAGE_SIZE,
        signal: controller.signal,
      })
      .then((response) => {
        if (controller.signal.aborted) return;

        setIncidents(
          Array.isArray(response.items) ? response.items : []
        );
        setTotal(response.total);

        const lastPage = Math.max(
          0,
          Math.ceil(response.total / PAGE_SIZE) - 1
        );

        if (page > lastPage) {
          setPage(lastPage);
        }
      })
      .catch((requestError) => {
        if (!controller.signal.aborted) {
          setListError(getApiErrorMessage(requestError));
        }
      })
      .finally(() => {
        if (!controller.signal.aborted) {
          setLoadingIncidents(false);
        }
      });

    return () => controller.abort();
  }, [search, page, refresh]);

  function handleSearch(event) {
    event.preventDefault();

    setPage(0);
    setSearch(searchInput.trim());
    setRefresh((value) => value + 1);
  }

  function toggleProvider(providerId) {
    if (
      running ||
      !privacyPolicy ||
      !allowedProviders.includes(providerId)
    ) {
      return;
    }

    setSelectedProviders((previous) =>
      previous.includes(providerId)
        ? previous.filter((value) => value !== providerId)
        : [...previous, providerId]
    );
  }

  async function runEvaluation() {
    if (inFlight.current) return;

    setError("");

    if (!privacyPolicy) {
      setError(
        "Load the workspace privacy policy before running a comparison."
      );
      return;
    }

    if (!selectedIncident || effectiveProviders.length < 2) {
      setError(
        "Select an incident and at least two allowed providers."
      );
      return;
    }

    inFlight.current = true;
    setRunning(true);
    setEvaluation(null);

    const incidentSnapshot = selectedIncident;
    const providerSnapshot = [...effectiveProviders];

    try {
      const response = await evaluationService.run({
        incidentId: incidentSnapshot.id,
        providers: providerSnapshot,
      });

      if (
        response?.incident_id !== incidentSnapshot.id ||
        !Array.isArray(response.results)
      ) {
        setError(
          "The API returned an unexpected evaluation response."
        );
        return;
      }

      setEvaluation({
        ...response,
        incidentTitle: incidentSnapshot.title,
      });
    } catch (requestError) {
      setError(getApiErrorMessage(requestError));
    } finally {
      inFlight.current = false;
      setRunning(false);
    }
  }

  return (
    <>
      <PrivacyControl
        onPolicyChange={setPrivacyPolicy}
        disabled={running}
      />

      <div className="mt-8 grid items-start gap-6 xl:grid-cols-2">
        <section className="min-w-0 rounded-lg border border-line bg-surface p-5">
          <h2 className="text-sm font-semibold">
            Select an incident
          </h2>

          <form
            onSubmit={handleSearch}
            className="mt-4 flex gap-2"
          >
            <input
              type="search"
              aria-label="Search incidents for comparison"
              placeholder="Search title or filename"
              value={searchInput}
              onChange={(event) =>
                setSearchInput(event.target.value)
              }
              disabled={running}
              className="h-10 min-w-0 flex-1 rounded-md border border-line bg-canvas px-3 text-sm outline-none focus:border-accent disabled:opacity-50"
            />

            <button
              type="submit"
              disabled={running}
              aria-label="Search incidents"
              className="flex h-10 w-10 shrink-0 items-center justify-center rounded-md border border-line hover:bg-elevated focus-visible:outline focus-visible:outline-2 focus-visible:outline-accent disabled:opacity-50"
            >
              <Search size={17} aria-hidden="true" />
            </button>
          </form>

          {selectedIncident && (
            <div className="mt-4 rounded-md border border-accent/20 bg-accent/5 px-3 py-2">
              <p className="break-words text-xs leading-6 text-accent">
                Selected: {selectedIncident.title}
              </p>

              <p className="break-all font-mono text-[11px] leading-5 text-text-muted">
                {selectedIncident.id}
              </p>
            </div>
          )}

          {loadingIncidents ? (
            <div
              role="status"
              className="mt-5 flex items-center gap-2 text-sm text-text-secondary"
            >
              <LoaderCircle
                size={16}
                className="animate-spin"
                aria-hidden="true"
              />
              Loading incidents…
            </div>
          ) : listError ? (
            <div className="mt-5">
              <p role="alert" className="text-sm text-red-300">
                {listError}
              </p>

              <button
                type="button"
                onClick={() =>
                  setRefresh((value) => value + 1)
                }
                disabled={running}
                className="mt-3 text-sm text-accent hover:underline disabled:opacity-50"
              >
                Retry
              </button>
            </div>
          ) : (
            <>
              <fieldset disabled={running} className="mt-4">
                <legend className="sr-only">
                  Available incidents
                </legend>

                <div className="technical-scroll max-h-64 divide-y divide-line overflow-y-auto">

                          setSelectedIncident(incident)

import axios from "axios";

const configuredBaseURL = import.meta.env.VITE_API_BASE_URL;

if (!configuredBaseURL) {
  throw new Error(
    "VITE_API_BASE_URL is missing. Set it in frontend/.env and restart Vite."
  );
}

const api = axios.create({
  baseURL: configuredBaseURL.replace(/\/+$/, ""),
  timeout: 30000,
  headers: {
    Accept: "application/json",
  },
});

let accessToken = null;

export function setAccessToken(token) {
  accessToken = token || null;
}

api.interceptors.request.use((config) => {
  if (accessToken && !config.skipAuth) {
    config.headers.Authorization = `Bearer ${accessToken}`;
  }

  return config;
});

export function getApiErrorMessage(error) {
  if (axios.isCancel(error)) {
    return "The request was cancelled.";
  }

  const detail = error?.response?.data?.detail;

  if (typeof detail === "string") {
    return detail;
  }

  if (Array.isArray(detail)) {
    return detail
      .map((item) => {
        const field = Array.isArray(item.loc)
          ? item.loc
              .filter((part) => !["body", "query", "path"].includes(part))
              .join(".")
          : "";

        const message = item.msg || "Invalid value";

        return field ? `${field}: ${message}` : message;
      })
      .join("; ");
  }

  if (error?.code === "ECONNABORTED") {
    return "The request timed out. Check the backend and try again.";
  }

  if (!error?.response) {
    return "Unable to reach the API. Check that the backend is running and that the API URL and CORS configuration are correct.";
  }

  switch (error.response.status) {
    case 401:
      return "Authentication is required. Please sign in again.";
    case 403:
      return "You do not have permission to perform this action.";
    case 404:
      return "The requested resource was not found.";
    case 409:
      return "This request conflicts with an existing resource.";
    case 429:
      return "Too many requests. Please try again shortly.";
    default:
      return "The request failed. Please try again.";
  }
}

export default api;

import api from "./api";

const authService = {
  async register({ username, password }) {
    const response = await api.post(
      "/api/auth/register",
      {
        username,
        password,
      },
      {
        skipAuth: true,
      }
    );

    return response.data;
  },

  async login({ username, password }) {
    const form = new URLSearchParams();

    form.set("grant_type", "password");
    form.set("username", username);
    form.set("password", password);

    const response = await api.post("/api/auth/token", form, {
      skipAuth: true,
      headers: {
        "Content-Type": "application/x-www-form-urlencoded",
      },
    });

    return response.data;
  },

  async getCurrentUser({ signal } = {}) {
    const response = await api.get("/api/auth/me", {
      signal,
    });

    return response.data;
  },

  async getUsers({ signal } = {}) {
    const response = await api.get("/api/auth/users", {
      signal,
    });

    return response.data;
  },
};
